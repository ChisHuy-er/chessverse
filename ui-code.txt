  useEffect(() => {

      setWelcomeToastId(0);

      window.clearTimeout(timer);

    try {
      const response = await fetch("/api/auth/me", {
        cache: "no-store",
      });

      if (!response.ok) return;

      const data = await response.json();

      if (active && data.user) {
        setUser(data.user);
      }
    } catch (error) {
      console.error("Không thể tải thông tin người dùng:", error);
    }
  }

  loadUser();

  return () => {
    active = false;
  };
  }, []);


  useEffect(() => {

    async function loadHistory() {
      try {
        const response = await fetch("/api/online/history", {
          cache: "no-store",
        });

        if (!response.ok) return;

        const data = await response.json();

        if (active && Array.isArray(data.matches)) {
          setHistoryMatches(data.matches);
        }
      } catch (error) {
        console.error("Không thể tải lịch sử đấu:", error);
      }
    }

    loadHistory();

    return () => {
      active = false;
    };

    function startMatch(mode: "ranked" | "casual" | "ai") {
        setModal(null);

          setPendingMode(mode);
          setAuthMode("login");
          setAuthOpen(true);
          setNotice("Bạn cần đăng nhập để bắt đầu trận đấu.");
          return;
        }

        if (mode === "ranked" || mode === "casual") {
          setOnlineMode(mode);
          setOnlineOpen(true);
        } else {
          router.push("/play");
        }
      }

      function handleAuthSuccess(profile: UserProfile) {
        setUser(profile);
        setAuthOpen(false);
        setNotice("");
        setWelcomeToastId((current) => current + 1);

        if (pendingMode) {
          const selectedMode = pendingMode;
          setPendingMode(null);

          if (selectedMode === "ranked" || selectedMode === "casual") {
            setOnlineMode(selectedMode);
            setOnlineOpen(true);
          } else {
            router.push("/play");
          }
        }
      }

      async function handleLogout() {
        try {

  } catch (error) {
    console.error("Đăng xuất thất bại:", error);
    setNotice("Không thể đăng xuất. Vui lòng thử lại.");

          {welcomeToastId > 0 && (
      <div
        className="welcome-toast"
        role="status"
        aria-live="polite"
      >
        <div className="welcome-toast-icon">
          ♞
        </div>

        <div className="welcome-toast-content">
          <span className="welcome-toast-brand">
            CHESSVERSE
          </span>

          <strong>Đăng nhập thành công</strong>

          <p>
            Chào mừng bạn đến với CHESSVERSE!
          </p>

          <small>
            Rất vui khi bạn tham gia. Hãy sẵn sàng cho
            những trận đấu cờ vua đầy thử thách!
          </small>
        </div>

        <button
          type="button"
          aria-label="Đóng thông báo"
          onClick={() => setWelcomeToastId(0)}
        >
          ×
        </button>
      </div>
    )}
      {onlineOpen && (
      <OnlineMatchOverlay
        mode={onlineMode}
        onClose={() => setOnlineOpen(false)}
      />
      )}

    <AuthModal
      key={authMode}
      open={authOpen}
      initialMode={authMode}
      onClose={() => {
        setAuthOpen(false);
        setPendingMode(null);
      }}
      onSuccess={handleAuthSuccess}
    />
    </main>
  );
}

"use client";

import { useState } from "react";
import type { FormEvent } from "react";

export type UserProfile = {
  id: string;
  email: string;
  username: string;
  displayName: string;
  avatarUrl?: string | null;
  elo: number;
  createdAt?: string;
};

type AuthModalProps = {
  open: boolean;
  initialMode: "login" | "register";
  onClose: () => void;
  onSuccess: (user: UserProfile) => void;
};

export default function AuthModal({
  open,
  initialMode,
  onClose,
  onSuccess,
}: AuthModalProps) {
  const [mode, setMode] = useState<"login" | "register">(initialMode);
  const [identifier, setIdentifier] = useState("");
  const [email, setEmail] = useState("");
  const [username, setUsername] = useState("");
  const [displayName, setDisplayName] = useState("");
  const [password, setPassword] = useState("");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  if (!open) return null;

  function switchMode(nextMode: "login" | "register") {
    setMode(nextMode);
    setError("");
    setPassword("");
  }

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setError("");
    setLoading(true);

    try {
      const endpoint =
        mode === "login"
          ? "/api/auth/login"
          : "/api/auth/register";

      const body =
        mode === "login"
          ? { identifier, password }
          : { email, username, password, displayName };

      const response = await fetch(endpoint, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify(body),
      });

      const data = await response.json();

      if (!response.ok) {
        const message = Array.isArray(data.message)
          ? data.message.join(", ")
          : data.message;

        throw new Error(message || "Đã xảy ra lỗi. Vui lòng thử lại.");
      }

      if (!data.user) {
        throw new Error("Máy chủ không trả về thông tin người chơi.");
      }

      onSuccess(data.user as UserProfile);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Không thể kết nối đến máy chủ.",
      );
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="modal-backdrop auth-backdrop" onClick={onClose}>
      <div

              />
            </>
          ) : (
            <>
              <label htmlFor="auth-identifier">
                Email hoặc Username
              </label>
              <input
                id="auth-identifier"
                type="text"
                placeholder="Nhập email hoặc username"
                value={identifier}
                onChange={(event) => setIdentifier(event.target.value)}
                minLength={3}
                maxLength={255}
                required
              />
            </>
          )}

          <label htmlFor="auth-password">Mật khẩu</label>
          <input
            id="auth-password"
            type="password"
            placeholder="Ít nhất 8 ký tự"
            value={password}
            onChange={(event) => setPassword(event.target.value)}
            minLength={8}
            maxLength={72}
            required
          />

          {error && (
            <div className="auth-error">
              {error}
            </div>
          )}

          <button
            type="submit"
            className="primary-button auth-submit"
            disabled={loading}
          >
            {loading
              ? "ĐANG XỬ LÝ..."
              : mode === "login"
                ? "ĐĂNG NHẬP"
                : "TẠO TÀI KHOẢN"}
          </button>
        </form>

        <p className="auth-switch-text">
          {mode === "login"
            ? "Chưa có tài khoản?"
            : "Đã có tài khoản?"}{" "}
          <button
            type="button"
            onClick={() =>
              switchMode(mode === "login" ? "register" : "login")
            }
          >
            {mode === "login" ? "Đăng ký ngay" : "Đăng nhập"}
          </button>
        </p>
      </div>
    </div>
  );
}

"use client";

import {
  createContext,
  useContext,
  useEffect,
  useSyncExternalStore,
  type ReactNode,
} from "react";

type Theme = "dark" | "light";

type ThemeContextType = {
  theme: Theme;
  toggleTheme: () => void;
};

const STORAGE_KEY = "chessverse-theme";
const THEME_EVENT = "chessverse-theme-change";

const ThemeContext = createContext<ThemeContextType | null>(null);

// Đọc theme đã lưu trong trình duyệt
function getThemeSnapshot(): Theme {
  return localStorage.getItem(STORAGE_KEY) === "light"
    ? "light"
    : "dark";
}

// Theme mặc định khi render phía server
function getServerThemeSnapshot(): Theme {
  return "dark";
}

// Theo dõi thay đổi theme
function subscribeToTheme(callback: () => void) {
  const handleChange = () => {
    callback();
  };

  window.addEventListener("storage", handleChange);
  window.addEventListener(THEME_EVENT, handleChange);

  return () => {
    window.removeEventListener("storage", handleChange);
    window.removeEventListener(THEME_EVENT, handleChange);
  };
}

export function ThemeProvider({
  children,
}: {
  children: ReactNode;
}) {
  const theme = useSyncExternalStore(
    subscribeToTheme,
    getThemeSnapshot,
    getServerThemeSnapshot,
  );

  // Đồng bộ theme với HTML
  useEffect(() => {
    document.documentElement.dataset.theme = theme;
  }, [theme]);

  function toggleTheme() {
    const nextTheme: Theme =
      theme === "dark" ? "light" : "dark";

    localStorage.setItem(STORAGE_KEY, nextTheme);
    document.documentElement.dataset.theme = nextTheme;

    // Thông báo cho các component và tab đang theo dõi
    window.dispatchEvent(new Event(THEME_EVENT));
  }

  return (
    <ThemeContext.Provider
      value={{
        theme,
        toggleTheme,
      }}
    >
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  const context = useContext(ThemeContext);

  if (!context) {
    throw new Error(
      "useTheme phải được sử dụng bên trong ThemeProvider",
    );
  }

  return context;
}
